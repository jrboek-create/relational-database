Set-StrictMode -Version Latest
$ErrorActionPreference = 'Stop'

Set-Location -LiteralPath 'C:\Directus\relational-repository'

$Source = 'C:\Directus\relational-repository\imports_hold\ER_SR_1CAT_UNIV_MASTER_DETAIL_618456250.xls'
$ExpectedBytes = [int64]7435264
$ExpectedHash = '91FEA403B0BCCA8CC746EC28723C576A4D05204E1F1C9D8D7D16D451DB2A7D09'
$TargetTable = 'staging.raw_er_sr_1cat_univ_master_detail_618456250_s01_xls_91fea403'
$ImportBatchId = [guid]::NewGuid().ToString()
$ImportedAtUtc = [datetime]::UtcNow.ToString('yyyy-MM-ddTHH:mm:ss.ffffffZ')

$Excel = $null
$Book = $null
$Sheet = $null
$Range = $null

function ConvertTo-CsvField {
    param([AllowNull()][object]$Value)
    if ($null -eq $Value) { return '""' }
    $Text = ([string]$Value).Replace('"','""')
    return '"' + $Text + '"'
}

try {
    Write-Host ''
    Write-Host '=== DEPARTMENT STAGING CONTROLLED PERSISTENT LOAD ==='

    $File = Get-Item -LiteralPath $Source -ErrorAction Stop
    $Hash = (Get-FileHash -LiteralPath $Source -Algorithm SHA256).Hash.ToUpperInvariant()

    if ($File.Length -ne $ExpectedBytes) { throw 'SOURCE_BYTE_COUNT_MISMATCH' }
    if ($Hash -ne $ExpectedHash) { throw 'SOURCE_SHA256_MISMATCH' }

    Write-Host "SOURCE_BYTES=$($File.Length)"
    Write-Host "SOURCE_SHA256=$Hash"
    Write-Host "IMPORT_BATCH_ID=$ImportBatchId"

    $Excel = New-Object -ComObject Excel.Application
    $Excel.Visible = $false
    $Excel.DisplayAlerts = $false
    $Excel.AskToUpdateLinks = $false
    $Excel.AutomationSecurity = 3

    $Book = $Excel.Workbooks.Open($Source, 0, $true)
    $Sheet = $Book.Worksheets.Item(1)
    $Range = $Sheet.UsedRange
    $Values = $Range.Value2

    if (
        ([string]$Values.GetValue(2,1)).Trim() -ne 'Univ Course ID' -or
        ([string]$Values.GetValue(2,2)).Trim() -ne 'Eff Date' -or
        ([string]$Values.GetValue(2,50)).Trim() -ne 'WW Owner(s)' -or
        ([string]$Values.GetValue(2,51)).Trim() -ne 'WW College(s)'
    ) { throw 'SOURCE_HEADER_MISMATCH' }

    $Sql = New-Object System.Text.StringBuilder
    [void]$Sql.AppendLine('\set ON_ERROR_STOP on')
    [void]$Sql.AppendLine('\pset pager off')
    [void]$Sql.AppendLine('BEGIN;')
    [void]$Sql.AppendLine(@'
CREATE TABLE staging.raw_er_sr_1cat_univ_master_detail_618456250_s01_xls_91fea403 (
    source_row_number integer NOT NULL,
    university_course_id text NOT NULL,
    effective_date date NOT NULL,
    ww_owner_raw text,
    ww_college_raw text,
    source_filename text NOT NULL,
    source_sheet text NOT NULL,
    source_header_row integer NOT NULL,
    source_sha256 text NOT NULL,
    source_bytes bigint NOT NULL,
    import_batch_id uuid NOT NULL,
    imported_at_utc timestamptz NOT NULL,
    CONSTRAINT raw_er_sr_1cat_univ_master_detail_618456250_s01_xls_91fea403_row_uq
        UNIQUE (source_row_number),
    CONSTRAINT raw_er_sr_1cat_univ_master_detail_618456250_s01_xls_91fea403_id_uq
        UNIQUE (university_course_id)
);

COPY staging.raw_er_sr_1cat_univ_master_detail_618456250_s01_xls_91fea403 (
    source_row_number,
    university_course_id,
    effective_date,
    ww_owner_raw,
    ww_college_raw,
    source_filename,
    source_sheet,
    source_header_row,
    source_sha256,
    source_bytes,
    import_batch_id,
    imported_at_utc
)
FROM STDIN
WITH (FORMAT csv);
'@)

    $SourceRows = 0
    for ($Row = 3; $Row -le [int]$Range.Rows.Count; $Row++) {
        $RawId = $Values.GetValue($Row,1)
        if ($null -eq $RawId) { continue }

        if ($RawId -is [double]) { $Id = ([int64]$RawId).ToString() }
        else { $Id = ([string]$RawId).Trim() }
        if ($Id.Length -eq 0) { continue }

        $RawDate = $Values.GetValue($Row,2)
        if ($RawDate -is [double]) {
            $EffectiveDate = ([datetime]::FromOADate([double]$RawDate)).ToString('yyyy-MM-dd')
        }
        else {
            $ParsedDate = [datetime]::MinValue
            if (-not [datetime]::TryParse(([string]$RawDate), [ref]$ParsedDate)) {
                throw "INVALID_EFFECTIVE_DATE_AT_SOURCE_ROW=$Row"
            }
            $EffectiveDate = $ParsedDate.ToString('yyyy-MM-dd')
        }

        $Department = ''
        if ($null -ne $Values.GetValue($Row,50)) {
            $Department = ([string]$Values.GetValue($Row,50)).Trim()
        }

        $College = ''
        if ($null -ne $Values.GetValue($Row,51)) {
            $College = ([string]$Values.GetValue($Row,51)).Trim()
        }

        $CsvLine = @(
            (ConvertTo-CsvField $Row),
            (ConvertTo-CsvField $Id),
            (ConvertTo-CsvField $EffectiveDate),
            (ConvertTo-CsvField $Department),
            (ConvertTo-CsvField $College),
            (ConvertTo-CsvField $File.Name),
            (ConvertTo-CsvField $Sheet.Name),
            (ConvertTo-CsvField 2),
            (ConvertTo-CsvField $Hash),
            (ConvertTo-CsvField $File.Length),
            (ConvertTo-CsvField $ImportBatchId),
            (ConvertTo-CsvField $ImportedAtUtc)
        ) -join ','

        [void]$Sql.AppendLine($CsvLine)
        $SourceRows++
    }

    if ($SourceRows -ne 2670) { throw "SOURCE_ROWS_PREPARED_MISMATCH=$SourceRows" }
    [void]$Sql.AppendLine('\.')

    [void]$Sql.AppendLine(@'
DO $assert$
DECLARE
    v_loaded_rows bigint;
    v_distinct_ids bigint;
    v_duplicates bigint;
    v_catalog_rows bigint;
    v_department_matched bigint;
    v_department_unmatched bigint;
    v_college_matched bigint;
    v_college_unmatched bigint;
    v_combined_matched bigint;
    v_ambiguous bigint;
    v_college_agreement bigint;
    v_college_disagreement bigint;
    v_lineage_rows bigint;
BEGIN
    SELECT COUNT(*), COUNT(DISTINCT university_course_id)
    INTO v_loaded_rows, v_distinct_ids
    FROM staging.raw_er_sr_1cat_univ_master_detail_618456250_s01_xls_91fea403;

    SELECT COUNT(*) INTO v_duplicates
    FROM (
        SELECT university_course_id
        FROM staging.raw_er_sr_1cat_univ_master_detail_618456250_s01_xls_91fea403
        GROUP BY university_course_id
        HAVING COUNT(*) > 1
    ) d;

    WITH latest_date AS (
        SELECT university_course_id, MAX(effective_date) AS effective_date
        FROM staging.raw_er_sr_1cat_univ_master_detail_618456250_s01_xls_91fea403
        GROUP BY university_course_id
    ), latest AS (
        SELECT s.university_course_id, s.ww_owner_raw, s.ww_college_raw
        FROM staging.raw_er_sr_1cat_univ_master_detail_618456250_s01_xls_91fea403 s
        JOIN latest_date l
          ON l.university_course_id = s.university_course_id
         AND l.effective_date = s.effective_date
    ), rollup AS (
        SELECT
            university_course_id,
            MIN(NULLIF(BTRIM(ww_owner_raw),'')) AS department_name,
            MIN(NULLIF(BTRIM(ww_college_raw),'')) AS college_name,
            COUNT(DISTINCT ROW(
                NULLIF(BTRIM(ww_owner_raw),''),
                NULLIF(BTRIM(ww_college_raw),'')
            )) AS mapping_count
        FROM latest
        GROUP BY university_course_id
    ), proof AS (
        SELECT c.university_course_id, r.department_name, r.college_name, r.mapping_count
        FROM reporting.course_current_catalog c
        LEFT JOIN rollup r
          ON r.university_course_id = BTRIM(c.university_course_id)
    )
    SELECT
        COUNT(*),
        COUNT(*) FILTER (WHERE mapping_count = 1 AND department_name IS NOT NULL),
        COUNT(*) FILTER (WHERE department_name IS NULL),
        COUNT(*) FILTER (WHERE mapping_count = 1 AND college_name IS NOT NULL),
        COUNT(*) FILTER (WHERE college_name IS NULL),
        COUNT(*) FILTER (
            WHERE mapping_count = 1

    INTO v_college_agreement, v_college_disagreement
    FROM reporting.course_catalog_detail d
    JOIN workbook_college w
      ON w.university_course_id = BTRIM(d.university_course_id)
    WHERE NULLIF(BTRIM(d.college_name),'') IS NOT NULL
      AND w.college_name IS NOT NULL;

    SELECT COUNT(*) INTO v_lineage_rows
    FROM staging.raw_er_sr_1cat_univ_master_detail_618456250_s01_xls_91fea403
    WHERE source_row_number IS NOT NULL
      AND source_filename = 'ER_SR_1CAT_UNIV_MASTER_DETAIL_618456250.xls'
      AND source_sheet = 'sheet1'
      AND source_header_row = 2
      AND source_sha256 = '91FEA403B0BCCA8CC746EC28723C576A4D05204E1F1C9D8D7D16D451DB2A7D09'
      AND source_bytes = 7435264
      AND import_batch_id IS NOT NULL
      AND imported_at_utc IS NOT NULL;

    IF v_loaded_rows <> 2670 THEN RAISE EXCEPTION 'loaded_rows=% expected=2670', v_loaded_rows; END IF;
    IF v_distinct_ids <> 2670 THEN RAISE EXCEPTION 'distinct_ids=% expected=2670', v_distinct_ids; END IF;
    IF v_duplicates <> 0 THEN RAISE EXCEPTION 'duplicates=% expected=0', v_duplicates; END IF;
    IF v_catalog_rows <> 966 THEN RAISE EXCEPTION 'catalog_rows=% expected=966', v_catalog_rows; END IF;
    IF v_department_matched <> 964 THEN RAISE EXCEPTION 'department_matched=% expected=964', v_department_matched; END IF;
    IF v_department_unmatched <> 2 THEN RAISE EXCEPTION 'department_unmatched=% expected=2', v_department_unmatched; END IF;
    IF v_college_matched <> 964 THEN RAISE EXCEPTION 'college_matched=% expected=964', v_college_matched; END IF;
    IF v_college_unmatched <> 2 THEN RAISE EXCEPTION 'college_unmatched=% expected=2', v_college_unmatched; END IF;
    IF v_combined_matched <> 964 THEN RAISE EXCEPTION 'combined_matched=% expected=964', v_combined_matched; END IF;
    IF v_ambiguous <> 0 THEN RAISE EXCEPTION 'ambiguous=% expected=0', v_ambiguous; END IF;
    IF v_college_agreement <> 964 THEN RAISE EXCEPTION 'college_agreement=% expected=964', v_college_agreement; END IF;
    IF v_college_disagreement <> 0 THEN RAISE EXCEPTION 'college_disagreement=% expected=0', v_college_disagreement; END IF;
    IF v_lineage_rows <> 2670 THEN RAISE EXCEPTION 'lineage_rows=% expected=2670', v_lineage_rows; END IF;

$assert$;

SELECT 'loaded_rows', COUNT(*)::text
FROM staging.raw_er_sr_1cat_univ_master_detail_618456250_s01_xls_91fea403;

FROM staging.raw_er_sr_1cat_univ_master_detail_618456250_s01_xls_91fea403;
SELECT 'duplicates', COUNT(*)::text
FROM (
    SELECT university_course_id
    FROM staging.raw_er_sr_1cat_univ_master_detail_618456250_s01_xls_91fea403
    GROUP BY university_course_id HAVING COUNT(*) > 1
) d;
SELECT 'lineage_rows', COUNT(*)::text
FROM staging.raw_er_sr_1cat_univ_master_detail_618456250_s01_xls_91fea403
WHERE source_sha256 = '91FEA403B0BCCA8CC746EC28723C576A4D05204E1F1C9D8D7D16D451DB2A7D09';

COMMIT;

SELECT 'TARGET_TABLE_EXISTS',
       CASE WHEN to_regclass('staging.raw_er_sr_1cat_univ_master_detail_618456250_s01_xls_91fea403') IS NULL
            THEN 'FALSE' ELSE 'TRUE' END;
'@)

    Write-Host "SOURCE_ROWS_PREPARED=$SourceRows"
    Write-Host ''
    Write-Host '=== POSTGRESQL CONTROLLED PERSISTENT LOAD ==='

    $OldPreference = $ErrorActionPreference
    $ErrorActionPreference = 'Continue'
    try {
        $Output = @(
            $Sql.ToString() |
                docker compose exec -T database `
                    psql -X -v ON_ERROR_STOP=1 -P pager=off -A -t -F '|' `
                    -U directus_repository -d directus_repository 2>&1
        )
        $ExitCode = $LASTEXITCODE
    }
    finally {
        $ErrorActionPreference = $OldPreference
    }

    $Output | Where-Object { $_ -and $_.Trim().Length -gt 0 } | ForEach-Object { Write-Host $_ }

    if ($ExitCode -ne 0) {
        throw ('CONTROLLED_PERSISTENT_LOAD_FAILED: ' + ($Output -join [Environment]::NewLine))
    }

    if (-not ($Output -contains 'TARGET_TABLE_EXISTS|TRUE')) {
        throw 'POST_COMMIT_TARGET_TABLE_VERIFICATION_FAILED'
    }

    Write-Host ''
    Write-Host 'PASS|DEPARTMENT_STAGING_CONTROLLED_PERSISTENT_LOAD'
    Write-Host 'PERSISTENT_DATABASE_MUTATED=TRUE'
    Write-Host 'WORKBOOK_MODIFIED=FALSE'
    Write-Host 'DIRECTUS_MUTATED=FALSE'
    Write-Host 'MIDDLEWARE_MUTATED=FALSE'
    Write-Host 'UI_MUTATED=FALSE'
}
finally {
    if ($null -ne $Book) { try { $Book.Close($false) } catch {} }
    if ($null -ne $Excel) { try { $Excel.Quit() } catch {} }

    foreach ($Object in @($Range,$Sheet,$Book,$Excel)) {
        if ($null -ne $Object -and [Runtime.InteropServices.Marshal]::IsComObject($Object)) {
            try { [void][Runtime.InteropServices.Marshal]::FinalReleaseComObject($Object) } catch {}
        }
    }

    [GC]::Collect()
    [GC]::WaitForPendingFinalizers()
}
